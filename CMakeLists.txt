cmake_minimum_required(VERSION 3.16)
project(StageSignal VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SS_BASE_X86 "Legacy baseline: i586 codegen, no SSE/SIMD (spec §6)" OFF)
option(SS_BUILD_TESTS "Build portable unit tests" ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(MSVC)
  add_compile_options(/W3 /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wno-unused-parameter -O2)
  if(SS_BASE_X86 AND CMAKE_SIZEOF_VOID_P EQUAL 4)
    add_compile_options(-march=i586 -mtune=generic -mno-sse -mno-mmx -mfpmath=387)
  endif()
endif()
include_directories(${CMAKE_SOURCE_DIR}/src ${CMAKE_SOURCE_DIR}/third_party)
find_package(Threads REQUIRED)

# One static library per portable module directory (all *.cpp below it, recursive).
set(SS_PORTABLE core render media midi lighting recording broadcast persistence diagnostics perf)
set(SS_LIBS "")
foreach(m ${SS_PORTABLE})
  file(GLOB_RECURSE srcs CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/${m}/*.cpp ${CMAKE_SOURCE_DIR}/src/${m}/*.c)
  if(srcs)
    add_library(ss_${m} STATIC ${srcs})
    target_link_libraries(ss_${m} PUBLIC Threads::Threads)
    list(APPEND SS_LIBS ss_${m})
  endif()
endforeach()
# Portable libs may reference each other; link as a group.
set(SS_ALL ${SS_LIBS})

if(SS_BUILD_TESTS)
  enable_testing()
  file(GLOB tests CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/tests/test_*.cpp)
  foreach(t ${tests})
    get_filename_component(n ${t} NAME_WE)
    add_executable(${n} ${t})
    if(UNIX)
      target_link_libraries(${n} -Wl,--start-group ${SS_ALL} -Wl,--end-group)
    else()
      target_link_libraries(${n} ${SS_ALL} ${SS_ALL})
    endif()
    if(WIN32)
      target_link_libraries(${n} ws2_32 winmm)
    endif()
    add_test(NAME ${n} COMMAND ${n} WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
  endforeach()
endif()

if(WIN32)
  file(GLOB_RECURSE winsrcs CONFIGURE_DEPENDS src/platform/windows/*.cpp src/ui/*.cpp src/app/*.cpp)
  file(GLOB rc CONFIGURE_DEPENDS src/app/*.rc)
  if(winsrcs)
    add_executable(StageSignal WIN32 ${winsrcs} ${rc})
    target_link_libraries(StageSignal -Wl,--start-group ${SS_ALL} -Wl,--end-group
      ole32 oleaut32 uuid winmm ws2_32 user32 gdi32 comctl32 comdlg32 shell32 advapi32 crypt32
      mfplat mf mfreadwrite mfuuid shlwapi dwmapi uxtheme avrt iphlpapi psapi)
    target_link_options(StageSignal PRIVATE -static -mwindows)
    target_compile_definitions(StageSignal PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0601 WINVER=0x0601)
  endif()
endif()
